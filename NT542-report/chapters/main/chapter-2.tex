\section{Phương pháp tiếp cận}
\subsection{Manual Configuration}
\vspace{5pt}
\noindent Cấu hình thủ công là phương pháp truyền thống trong quản trị hệ thống. Administrator trực tiếp thực hiện các thay đổi trên hệ thống thông qua command line, giao diện đồ họa hoặc file config.

\vspace{5pt}
\noindent \textbf{Các công cụ sử dụng trong cấu hình thủ công:}
\begin{itemize}
\item \textbf{Command Line Tools:} \texttt{ansible-playbook}, \texttt{systemctl}, \texttt{ufw}, \texttt{iptables}, \texttt{nftables},\\
\texttt{chmod}, \texttt{chown}, \texttt{mount}, \texttt{netstat}, \texttt{lynis}, \texttt{aide}, \texttt{auditd}, \texttt{fail2ban}
\item \textbf{Giao diện đồ họa (GUI):} AWX/Tower, Grafana, Webmin, Cockpit
\item \textbf{File cấu hình:} \texttt{/etc/sysctl.conf}, \texttt{/etc/fstab}, \texttt{/etc/ssh/sshd\_config},\\
\texttt{/etc/network/interfaces}, \texttt{/etc/passwd}, \texttt{/etc/shadow}, \texttt{ansible.cfg}, \texttt{inventory/hosts}, \texttt{playbooks/}
\end{itemize}

\vspace{5pt}
\noindent Phương pháp này có ưu điểm là linh hoạt và dễ kiểm soát từng bước, nhưng nhược điểm lớn là tốn nhiều thời gian, dễ xảy ra sai sót, và khó đảm bảo tính nhất quán giữa nhiều hệ thống khác nhau.

\vspace{5pt}
\noindent Đối với các hệ thống quy mô nhỏ, cấu hình thủ công vẫn có thể được áp dụng cho các tác vụ đặc thù yêu cầu sự can thiệp chuyên sâu.

\subsection{Automation bằng Ansible}
\vspace{5pt}
\noindent Ansible là công cụ tự động hóa mạnh mẽ giúp loại bỏ các thao tác thủ công lặp đi lặp lại. Với Ansible, chúng ta có thể định nghĩa toàn bộ quy trình hardening dưới dạng code, đảm bảo tính nhất quán và có thể tái sử dụng.

\vspace{5pt}
\noindent \textbf{Các công cụ sử dụng trong automation bằng Ansible:}
\begin{itemize}
\item \textbf{Ansible Command Line Tools:} \texttt{ansible-playbook}, \texttt{ansible-inventory},\\
\texttt{ansible-vault}, \texttt{ansible-galaxy}, \texttt{ansible-pull}
\item \textbf{Ansible GUI Tools:} AWX/Tower, Ansible Semaphore, Rundeck, Jenkins với Ansible plugin
\item \textbf{Ansible Configuration Files:} \texttt{ansible.cfg}, \texttt{inventory/hosts}, \texttt{playbooks/*.yml},\\
\texttt{roles/*/tasks/main.yml}, \texttt{group\_vars/*}, \texttt{host\_vars/*}, \texttt{vars/*}, \texttt{vault/*}
\end{itemize}

\vspace{5pt}
\noindent Ansible sử dụng YAML để định nghĩa các playbook, giúp administrator dễ dàng đọc, hiểu và chỉnh sửa. Khả năng idempotent của Ansible đảm bảo rằng việc chạy nhiều lần sẽ không gây ra các trạng thái không mong muốn.

\noindent
Hệ thống được triển khai theo mô hình Controller--Workers, trong đó một máy trung tâm (Controller) chịu trách nhiệm điều phối và quản lý cấu hình, trong khi các máy Worker đóng vai trò là các node mục tiêu được cấu hình tự động.

\vspace{5pt}
\noindent \textbf{Cụ thể trong đề tài:}
\begin{itemize}
\item \textbf{Controller:} Arch Linux (chạy Ansible)
\item \textbf{Worker 1:} Ubuntu 22.04
\item \textbf{Worker 2:} Ubuntu 24.04
\end{itemize}

\vspace{5pt}
\noindent Mô hình này cho phép thực hiện quản lý tập trung, đảm bảo tính nhất quán và giảm thiểu sai sót trong quá trình cấu hình hệ thống.

\subsection{Quy trình triển khai}

\noindent Quy trình triển khai hệ thống được thực hiện theo hướng tự động hóa thông qua Ansible, bao gồm các bước chính sau:

\vspace{5pt}
\noindent \textbf{1. Chuẩn bị môi trường}
\begin{itemize}
\item Máy Controller được cài đặt Ansible và các công cụ cần thiết
\item Các Worker Nodes được cài đặt SSH server và Python để hỗ trợ thực thi từ xa
\end{itemize}

\noindent \textbf{2. Thiết lập kết nối SSH}
\begin{itemize}
\item Cơ chế xác thực bằng SSH key được sử dụng để đảm bảo kết nối an toàn
\item Không cần nhập mật khẩu giữa Controller và các Worker
\end{itemize}

\noindent \textbf{3. Cấu hình Inventory}
\begin{itemize}
\item Danh sách các Worker Nodes được khai báo trong file inventory
\item Bao gồm địa chỉ IP, user và thông tin xác thực
\end{itemize}

\noindent \textbf{4. Triển khai cấu hình tự động}
\begin{itemize}
\item Các playbook Ansible được sử dụng để thực hiện cấu hình hệ thống
\item Áp dụng tiêu chuẩn CIS Benchmark trên toàn bộ các Worker
\end{itemize}

\noindent \textbf{5. Kiểm tra và đánh giá}
\begin{itemize}
\item Sau khi triển khai, hệ thống được kiểm tra lại thông qua các công cụ audit
\item Đánh giá mức độ tuân thủ bảo mật
\end{itemize}

\noindent \textbf{6. Bảo trì hệ thống}
\begin{itemize}
\item Hệ thống được duy trì thông qua việc audit định kỳ
\item Cập nhật playbook và giám sát log để phát hiện sớm các vấn đề
\end{itemize}

\vspace{5pt}
\noindent Quy trình này giúp đảm bảo việc triển khai hệ thống diễn ra nhất quán, có thể lặp lại và dễ dàng mở rộng.

\vspace{5pt}
\noindent \textbf{1. Auto-VM -- Tạo máy ảo tự động}

\noindent Thành phần này chịu trách nhiệm tự động hóa quá trình tạo và cấu hình máy ảo.

\vspace{5pt}
\noindent \textbf{Công nghệ:} KVM/QEMU, Ansible, Cloud-init

\noindent \textbf{Chức năng:}
\begin{itemize}
\item Tạo VM từ Ubuntu cloud image
\item Tự động cấu hình user, SSH key
\item Gán IP và cập nhật vào inventory
\end{itemize}

\noindent Đầu ra là các máy ảo sẵn sàng cho việc triển khai tiếp theo.

\vspace{5pt}
\noindent \textbf{2. CIS-Ubuntu -- Hardening bảo mật}

\noindent Thành phần này thực hiện triển khai tiêu chuẩn CIS Ubuntu 24.04 LTS Benchmark v1.0.0.

\noindent Hệ thống được chia thành 7 giai đoạn:
\begin{itemize}
\item Initial Setup
\item Services
\item Network
\item Firewall
\item Access Control
\item Logging
\item Maintenance
\end{itemize}

\noindent Mỗi giai đoạn được triển khai dưới dạng Ansible roles, đảm bảo:
\begin{itemize}
\item Tính modular
\item Khả năng tái sử dụng
\item Idempotent (chạy lại không gây lỗi)
\end{itemize}

\vspace{5pt}
\noindent \textbf{3. Báo cáo và đánh giá}

\noindent Thành phần này thực hiện đánh giá mức độ bảo mật của hệ thống sau khi hardening.

\noindent \textbf{Công cụ:} Lynis, OpenSCAP

\noindent \textbf{Đầu ra:}
\begin{itemize}
\item Báo cáo HTML
\item Điểm compliance
\item Danh sách lỗ hổng và khuyến nghị
\end{itemize}

\vspace{5pt}
\noindent \textbf{Quy trình hoạt động tổng thể}

\noindent Hệ thống hoạt động theo chu trình khép kín:
\begin{itemize}
\item Tạo máy ảo bằng Auto-VM
\item Triển khai hardening bằng CIS-Ubuntu
\item Đánh giá bằng Lynis và OpenSCAP
\item Phản hồi kết quả để cải tiến hệ thống
\end{itemize}

\vspace{5pt}
\noindent \textbf{Ưu điểm của mô hình}
\begin{itemize}
\item Tự động hóa toàn bộ quy trình
\item Đảm bảo tính nhất quán giữa các hệ thống
\item Dễ dàng mở rộng quy mô
\item Hỗ trợ kiểm soát phiên bản thông qua Git
\item Tích hợp cơ chế phản hồi (feedback loop)
\end{itemize}
